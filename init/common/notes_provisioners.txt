notes on provisioners (vagrant, ansible, saltstack)

Vagrant
-------

# transfer /tmp/scripts.tar files to /root
# vagrant provision --provision-with xferscripts

# provision with vagrant (using runscript)
# usage:
#   env RUNSCRIPT_ARGS="script.sh arg0 argN" \
      vagrant provision --provision-with runscript
# example:
#   env RUNSCRIPT_ARGS="upgradepkgs.sh" \
#     vagrant provision --provision-with runscript

# create Vagrant box from VM with vagrant user:
# example: (image: build/guest01/guest01.qcow2)
#   cd build/guest01
#   [env author=user01] sh vmrun.sh box_vagrant [guest01]
#
#   vagrant box add guest01_catalog.json
#   or
#   vagrant box add --provider libvirt --name user01/guest01 \
#     guest01-2020.09.10.libvirt.box

# init vagrant VM from box:
# example:
#   cd Documents/guest01
#   vagrant init [--minimal] user01/guest01
#   sed -i '/config.vm.box =/a \ \ config.vm.define \
#     ENV.fetch("vm_name", File.basename(Dir.pwd))' Vagrantfile
#   [env vm_name=guest01] vagrant up [--provider libvirt]


Ansible
-------

# provision with ansible ([check | run] tagged play(s)) [custom additional variables]
# usage:
#   ansible-playbook [-b][-k][-u usernm] [-i '10.0.2.10,' | -l 'guest#01,'] \
#     etc/ansible/playbook.yml [--check] [--extra-vars "var1=bsd1"] -t tag0[,tagN]
# example:
#   ansible-playbook [-b][-k][-u packer] -l 'guest#01,' \
#     etc/ansible/playbook.yml [--extra-vars "autoconfirm=YES"] -t upgradepkgs

# ad-hoc module execution ; ad-hoc shell cmd
# ansible [-b][-k][-u packer] -l 'guest#01,' all -m ping \
#   [-e 'ansible_python_interpreter="/usr/bin/env python"']
# ansible [-b][-k][-u packer] -l 'guest#01,' all -m package \
#   -a 'name="*" state=latest'
# ansible [-b][-k][-u packer] -l 'guest#01,' all -m shell \
#   -a 'pkg update ; pkg upgrade'

# ansible-doc
# usage:
#   ansible-doc [--help] --list [collection]
#   ansible-doc [-t module] <module>
# example(s):
#   ansible-doc --list ; ansible-doc [-t module] package service
#   ansible-doc -t callback default ; ansible-doc -t filter dirname

# edit/create ansible.cfg & etc/ansible/hosts
# ansible.cfg
# -------------------------------
# [defaults]
# inventory = etc/ansible/hosts
# #roles_path = etc/ansible/roles
# ##(note) interpreter_python = [auto[_silent] | '/usr/bin/env python']
# #interpreter_python = auto_silent
# display_skipped_hosts = False
# nocows = 1
# #timeout = 300
# callback_result_format = yaml

# #(note) ansible_python_interpreter: [auto[_silent] | '/usr/bin/env python']
# etc/ansible/hosts (yaml format)      etc/ansible/hosts (ini format)
# -----------------------------------  ------------------------------
# targets:                             [targets]
#   hosts:                             guest#01 \
#     guest#01:                          ansible_host=10.0.2.10 \
#       ansible_host: 10.0.2.10          ansible_python_interpreter=
#       ansible_python_interpreter: \      auto_silent
#         auto_silent                  [targets:vars]
#   vars:                              ansible_user=packer
#     ansible_user: packer             ansible_become=yes
#     ansible_become: yes

# how to bootstrap ansible role directory structure
# cd etc/ansible/roles ; ansible-galaxy init role1
# ansible-galaxy collection install [ansible.posix | community.general | l3d.avahi]

# syntax check
# ansible-playbook [-b][-k][-u packer] -l 'guest#01,' \
#   etc/ansible/playbook.yml --syntax-check -t upgradepkgs

# show available tags
# ansible-playbook [-b][-k][-u packer] -l 'guest#01,' \
#   etc/ansible/playbook.yml --list-tags --tags never,all

# show specified guest facts
# ansible [-b][-k][-u packer] -l 'guest#01,' all -m setup -a \
#   'gather_subset=!all,!min,kernel,os_family,lsb,distribution,service_mgr,pkg_mgr'


Saltstack
---------

# provision with salt ([test | apply] salt sls file(s)) [custom pillar values]
# usage:
#   salt-ssh [--sudo][--askpass][--user=usernm] \
#     ['*' | --roster=scan '10.0.2.10' | -L 'guest,'] state.apply sls0[,slsN] \
#     [test=True] [pillar='{"state1": {"var1": "bsd1"}}']
# example:
#   salt-ssh [--sudo][--askpass][--user=packer] -L 'guest#01,' state.apply \
#     upgradepkgs [pillar='{"state1": {"autoconfirm": "YES"}}']

# ad-hoc apply state ; ad-hoc shell cmd
# salt-ssh [--sudo][--askpass][--user=packer] -L 'guest#01,' test.ping
## salt-ssh [--sudo][--askpass][--user=packer] -L 'guest#01,' state.high \
##   '{"upgrade pkgs": {"pkg": ["uptodate"]}}'
# salt-ssh [--sudo][--askpass][--user=packer] -L 'guest#01,' state.single \
#   pkg.uptodate name='upgrade pkgs' [test=False] [refresh=False]
# salt-ssh [--sudo][--askpass][--user=packer] -L 'guest#01,' cmd.run \
#   'pkg update ; pkg upgrade'

# saltstack sys module
# usage:
#   salt-ssh ... sys.[state_]doc <module>[.<function>]
#   salt-ssh ... sys.list_[state_]modules
#   salt-ssh ... sys.list_[state_]functions <module>
# example(s):
#   salt-ssh ... sys.doc sys
#   salt-ssh ... sys.list_modules pkg service
#   salt-ssh ... sys.list_functions pkg service
#   salt-ssh ... sys.state_doc service.enabled
#   salt-ssh ... sys.doc pkg.install
